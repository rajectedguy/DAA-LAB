SIMPLE-FIND(x)

1. while parent[x] >= 0
2.     x = parent[x]

3. return x


SIMPLE-UNION(i, j)

1. root1 = SIMPLE-FIND(i)
2. root2 = SIMPLE-FIND(j)

3. if root1 != root2
4.     parent[root2] = root1



WEIGHTED-UNION(i, j)

1. root1 = FIND(i)
2. root2 = FIND(j)

3. if root1 == root2
4.     return

5. if parent[root1] <= parent[root2]
6.     parent[root1] = parent[root1] + parent[root2]
7.     parent[root2] = root1

8. else
9.     parent[root2] = parent[root1] + parent[root2]
10.    parent[root1] = root2





COLLAPSING-FIND(x)

1. root = x

2. while parent[root] >= 0
3.     root = parent[root]

4. while x != root
5.     parentNode = parent[x]
6.     parent[x] = root
7.     x = parentNode

8. return root
