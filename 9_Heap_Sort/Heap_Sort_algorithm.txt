HEAPSORT(arr, n)

1. Build MAX-HEAP(arr)

2. for i = n-1 down to 1
3.     swap arr[0] and arr[i]
4.     heapSize = heapSize - 1
5.     MAX-HEAPIFY(arr, 0)



MAX-HEAPIFY(arr, n, i)

1. largest = i
2. left = 2*i + 1
3. right = 2*i + 2

4. if left < n and arr[left] > arr[largest]
5.     largest = left

6. if right < n and arr[right] > arr[largest]
7.     largest = right

8. if largest != i
9.     swap arr[i] and arr[largest]
10.    MAX-HEAPIFY(arr, n, largest)
